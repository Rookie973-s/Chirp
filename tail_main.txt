            </div>
            <div style={{ display: 'flex', borderBottom: '1px solid var(--border-subtle)' }}>
              {['login', 'register'].map(tab => (
                <button key={tab} onClick={() => setAuthTab(tab)} style={{ flex: 1, padding: 16, fontSize: 15, fontWeight: 600, color: authTab === tab ? 'var(--brand-gold)' : 'var(--text-muted)', borderBottom: `2px solid ${authTab === tab ? 'var(--brand-gold)' : 'transparent'}`, textTransform: 'capitalize' }}>
                  {tab === 'login' ? 'Sign In' : 'Create Account'}
                </button>
              ))}
            </div>
            <div style={{ padding: '32px 24px' }}>
              <div style={{ display: 'flex', flexDirection: 'column', gap: 16 }}>
                {authTab === 'register' && (
                  <div style={{ display: 'flex', gap: 16 }}>
                    <input type="text" placeholder="First Name" className="input-field" />
                    <input type="text" placeholder="Last Name" className="input-field" />
                  </div>
                )}
                <input type="text" placeholder="Email or Matric Number" className="input-field" />
                <input type="password" placeholder="Password" className="input-field" />
                <button className="btn-primary" style={{ height: 48, fontSize: 16, marginTop: 8 }} onClick={() => { setIsLoggedIn(true); setAuthModalOpen(false); showToast('Welcome to ATBUConnect!', 'success'); }}>
                  {authTab === 'login' ? 'Sign In' : 'Create Account'}
                </button>
              </div>
            </div>
          </div>
        </div>

        {/* Staff Profile Modal */}
        {activeStaff && (
          <div style={{ position: 'fixed', inset: 0, background: 'rgba(6,18,32,0.6)', backdropFilter: 'blur(8px)', zIndex: 2000, display: staffModalOpen ? 'flex' : 'none', alignItems: 'center', justifyContent: 'center', opacity: staffModalOpen ? 1 : 0, transition: 'opacity 0.3s' }} onClick={(e) => e.target === e.currentTarget && setStaffModalOpen(false)}>
            <div style={{ background: 'var(--bg-surface)', width: '100%', maxWidth: 500, borderRadius: 'var(--radius-xl)', overflow: 'hidden', transform: staffModalOpen ? 'scale(1) translateY(0)' : 'scale(0.95) translateY(20px)', transition: 'all 0.4s cubic-bezier(0.34, 1.56, 0.64, 1)' }}>
              <div style={{ height: 100, background: `linear-gradient(135deg, ${activeStaff.bg}, var(--brand-navy))`, position: 'relative' }}>
                <button onClick={() => setStaffModalOpen(false)} style={{ position: 'absolute', top: 16, right: 16, color: '#fff', background: 'rgba(0,0,0,0.3)', borderRadius: '50%', padding: 8 }}><X size={16} /></button>
              </div>
              <div style={{ padding: '0 32px 32px', position: 'relative', marginTop: -40 }}>
                <div style={{ width: 80, height: 80, borderRadius: '50%', background: activeStaff.bg, border: '4px solid var(--bg-surface)', display: 'flex', alignItems: 'center', justifyContent: 'center', fontSize: 28, fontWeight: 700, color: '#fff', marginBottom: 16 }}>
                  {activeStaff.avatar}
                </div>
                <h2 style={{ fontSize: 22, fontWeight: 700, color: 'var(--text-main)', marginBottom: 4, display: 'flex', alignItems: 'center', gap: 8 }}>
                  {activeStaff.name} <CheckCircle size={18} color="var(--brand-gold)" />
                </h2>
                <p style={{ fontSize: 15, color: 'var(--text-muted)', marginBottom: 16 }}>{activeStaff.role} â€¢ {activeStaff.dept}</p>
                
                <div style={{ background: 'var(--bg-surface-hover)', borderRadius: 'var(--radius-md)', padding: 16, marginBottom: 24 }}>
                  <p style={{ fontSize: 14, color: 'var(--text-main)', lineHeight: 1.6, fontStyle: 'italic' }}>"{activeStaff.bio}"</p>
                </div>

                <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 16, marginBottom: 24 }}>
                  <div style={{ display: 'flex', alignItems: 'flex-start', gap: 12 }}>
                    <MapPin size={20} color="var(--brand-gold)" style={{ marginTop: 2 }} />
                    <div>
                      <span style={{ display: 'block', fontSize: 12, color: 'var(--text-faint)', fontWeight: 600 }}>OFFICE</span>
                      <span style={{ fontSize: 14, color: 'var(--text-main)' }}>{activeStaff.office}</span>
                    </div>
                  </div>
                  <div style={{ display: 'flex', alignItems: 'flex-start', gap: 12 }}>
                    <Clock size={20} color="var(--brand-gold)" style={{ marginTop: 2 }} />
                    <div>
                      <span style={{ display: 'block', fontSize: 12, color: 'var(--text-faint)', fontWeight: 600 }}>HOURS</span>
                      <span style={{ fontSize: 14, color: 'var(--text-main)' }}>{activeStaff.hours}</span>
                    </div>
                  </div>
                </div>

                <div style={{ display: 'flex', gap: 12 }}>
                  <button className="btn-primary" style={{ flex: 1 }} onClick={() => {
                    if(!isLoggedIn) { showToast('Sign in to message staff', 'error'); setStaffModalOpen(false); setAuthModalOpen(true); }
                    else { showToast('Message feature coming soon!', 'info'); }
                  }}>
                    <Mail size={18} /> Send Direct Message
                  </button>
                </div>
              </div>
            </div>
          </div>
        )}

        {/* Toasts */}
        <div style={{ position: 'fixed', bottom: 32, left: '50%', transform: 'translateX(-50%)', zIndex: 3000, display: 'flex', flexDirection: 'column', gap: 10, pointerEvents: 'none' }}>
          {toasts.map(t => (
            <div key={t.id} style={{ background: t.type === 'success' ? 'var(--status-success)' : t.type === 'error' ? 'var(--status-error)' : 'var(--text-main)', color: '#fff', padding: '12px 24px', borderRadius: 30, fontSize: 14, fontWeight: 600, display: 'flex', alignItems: 'center', gap: 10, boxShadow: 'var(--shadow-lg)', animation: 'popIn 0.3s var(--ease-spring)' }}>
              {t.type === 'success' && <CheckCircle size={18} />}
              {t.message}
            </div>
          ))}
        </div>
      </>
    );
  };

  return (
    <>
      <GlobalStyles />
      <Topbar />
      <div className="app-container">
        <LeftSidebar />
        <main style={{ minWidth: 0 }}>
          {/* Main Navigation Tabs */}
          {(activeView === 'feed' || activeView === 'staff') && (
            <div style={{ display: 'flex', gap: 8, marginBottom: 24, background: 'var(--bg-surface)', padding: 6, borderRadius: 'var(--radius-lg)', border: '1px solid var(--border-subtle)', boxShadow: 'var(--shadow-sm)' }}>
              <button onClick={() => handleNavigate('feed')} style={{ flex: 1, padding: 10, borderRadius: 'var(--radius-md)', fontSize: 14, fontWeight: 600, display: 'flex', alignItems: 'center', justifyContent: 'center', gap: 8, transition: 'all 0.2s', background: activeView === 'feed' ? (theme==='dark'?'var(--brand-gold)':'var(--brand-navy)') : 'transparent', color: activeView === 'feed' ? (theme==='dark'?'var(--brand-navy-dark)':'#fff') : 'var(--text-muted)' }}>
                <MessageSquare size={18} /> Question Feed
              </button>
              <button onClick={() => handleNavigate('staff')} style={{ flex: 1, padding: 10, borderRadius: 'var(--radius-md)', fontSize: 14, fontWeight: 600, display: 'flex', alignItems: 'center', justifyContent: 'center', gap: 8, transition: 'all 0.2s', background: activeView === 'staff' ? (theme==='dark'?'var(--brand-gold)':'var(--brand-navy)') : 'transparent', color: activeView === 'staff' ? (theme==='dark'?'var(--brand-navy-dark)':'#fff') : 'var(--text-muted)' }}>
                <Briefcase size={18} /> Staff Directory
              </button>
            </div>
          )}

          {activeView === 'feed' && <FeedView />}
          {activeView === 'staff' && <StaffView />}
          {activeView === 'detail' && <QuestionDetailView />}
          {activeView === 'apply' && <ApplyStaffView />}
          {activeView === 'profile' && <ProfileView />}
        </main>
        <RightSidebar />
      </div>
      <Modals />
    </>
  );
}
